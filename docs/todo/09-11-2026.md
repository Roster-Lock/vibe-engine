type EventListenerDefintion<T> = {
  event: string,
  sourceFilter?: (triggeringEntity: T)=>boolean
  sourceMaxMin: "*" | { min: number |"*", max: number | "*" },
  action: (triggeringEntities: Array<T>)=>any
}

type Game = {
  listeners: Record<string, ((triggeringEntities: Array<unknown>)=>any)>
  addToEventStack: (def: EventListenerDefintion<unknown>, triggeringEntities: Array<unknown>)=>any
}

function addGameEventListener<T>(game, eventListenerDef: EventListenerDefintion<T>){
  game.listeners[eventListenerDef.event] = (triggeringEntities: Array<T>)=>{
    const filteredEntities = filterEntities(triggeringEntities, def)
    if(filteredEntities === null) return;
    game.addToEventStack(def, filteredEntities)
  }
}

function filterEntities(triggeringEntities: Array<T>, def: EventListenerDefintion<T>){
  if(!def.sourceFilter) return triggeringEntities;
  if(def.sourceMaxMin === "*")
    return anyCount(triggeringEntities, def);
  if(def.sourceMaxMin.min === 1 && def.sourceMaxMin.max === "*")
    return anyCount(triggeringEntities, def);
  const filteredEntities = [];
  for(const entity of triggeringEntities){
    if(!def.sourceFilter(entity)) continue;
    filteredEntities.push(entity);
    if(def.sourceMaxMin.max === "*") continue;
    if(filteredEntities.length > def.sourceMaxMin.max) return null;
  }
  if(filteredEntities.length === 0) return null;
  if(def.sourceMaxMin.min === "*") return filteredEntities;
  if(filteredEntities.length < def.sourceMaxMin.min) retun null;
  return filteredEntities
}

function anyCount(triggeringEntities: Array<T>, def: EventListenerDefintion<T>){
  const filteredEntities = [];
  for(const entity of triggeringEntities){
    if(!def.sourceFilter(entity)) continue;
    filteredEntities.push(entity);
  }
  if(filteredEntities.length === 0) return null;
  return filteredEntities;
}
